        }
        let truncated = &desc[..MAX_DESC_LEN];

        }
    }

    **Highest severity helper** (used for host and service notes):
    ```rust
    fn highest_severity(vulns: &[Vulnerability]) -> Severity {
        vulns.iter()
            .filter_map(|v| v.cvss.as_ref())
            .map(|c| &c.severity)
            .max_by_key(|s| match s {
                Severity::Critical => 4, Severity::High => 3,
                Severity::Medium => 2, Severity::Low => 1, Severity::None => 0,
            })
            .cloned()
            .unwrap_or(Severity::None)
    }
    ```
